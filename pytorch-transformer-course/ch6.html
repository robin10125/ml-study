6.1 LayerNorm from scratch

PyTorch has nn.LayerNorm, but building it ourselves nails down two things: nn.Parameter (the learned scale and shift) and the biased-variance gotcha from Chapter 1.

class LayerNorm(nn.Module):
    def __init__(self, d_model, eps=1e-5):
        super().__init__()
        self.gamma = nn.Parameter(torch.ones(d_model))    # learned scale, starts at 1
        self.beta = nn.Parameter(torch.zeros(d_model))    # learned shift, starts at 0
        self.eps = eps

    def forward(self, x):                                 # x: (..., D)
        mean = x.mean(dim=-1, keepdim=True)               # (..., 1)
        var = x.var(dim=-1, keepdim=True, unbiased=False) # biased — matches nn.LayerNorm
        return (x - mean) / torch.sqrt(var + self.eps) * self.gamma + self.beta

Checklist: stats over dim=-1 (per token, over features — never the batch or time axis); keepdim=True so (...,1) broadcasts back over (...,D); eps inside the sqrt (else an all-constant token gives var=0 → division by zero → NaN); unbiased=False (§1.6); gamma/beta of shape (D,) as nn.Parameter so they're trained. gamma at ones and beta at zeros means the layer starts as the identity standardization.

6.2 CausalSelfAttention: the centerpiece

Transcribing §5.3 directly. Four nn.Linear projections (bias-free, like the original paper), the view/transpose head dance, batched scores, causal mask, softmax, and merge:

class CausalSelfAttention(nn.Module):
    def __init__(self, d_model, n_heads):
        super().__init__()
        assert d_model % n_heads == 0
        self.n_heads = n_heads
        self.d_head = d_model // n_heads
        self.wq = nn.Linear(d_model, d_model, bias=False)
        self.wk = nn.Linear(d_model, d_model, bias=False)
        self.wv = nn.Linear(d_model, d_model, bias=False)
        self.wo = nn.Linear(d_model, d_model, bias=False)

    def forward(self, x):                                # x: (B, T, D)
        B, T, D = x.shape
        H, hd = self.n_heads, self.d_head
        q = self.wq(x).view(B, T, H, hd).transpose(1, 2)  # (B, H, T, hd)
        k = self.wk(x).view(B, T, H, hd).transpose(1, 2)
        v = self.wv(x).view(B, T, H, hd).transpose(1, 2)

        scores = (q @ k.transpose(-2, -1)) / math.sqrt(hd)   # (B, H, T, T)
        mask = torch.tril(torch.ones(T, T, dtype=torch.bool, device=x.device))
        scores = scores.masked_fill(~mask, float("-inf"))    # before softmax!
        attn = F.softmax(scores, dim=-1)                     # (B, H, T, T)

        out = attn @ v                                       # (B, H, T, hd)
        out = out.transpose(1, 2).contiguous().view(B, T, D) # merge heads
        return self.wo(out)                                  # (B, T, D)

Why q @ k.transpose(-2, -1) computes everything at once: with heads on axis 1, the leading (B, H) are batch dimensions for @ (§4.4), and the matmul runs on the last two axes — a (T, hd) × (hd, T) per (batch, head) pair, i.e. all B·H score matrices in a single op, no Python loop. The mask is rebuilt each forward from T (you could cache it as a buffer — §3.4 — but rebuilding is simpler and the cost is trivial). Three classic bugs to not write: scaling by √D not √hd; masking after softmax; forgetting .contiguous() at the merge.

6.3 MLP and Block

class MLP(nn.Module):
    def __init__(self, d_model, d_ff):
        super().__init__()
        self.fc1 = nn.Linear(d_model, d_ff)
        self.fc2 = nn.Linear(d_ff, d_model)

    def forward(self, x):                                # (B,T,D) -> (B,T,D)
        return self.fc2(F.gelu(self.fc1(x)))             # widen, GELU, project back


class Block(nn.Module):
    def __init__(self, d_model, n_heads, d_ff):
        super().__init__()
        self.ln1 = LayerNorm(d_model)
        self.attn = CausalSelfAttention(d_model, n_heads)
        self.ln2 = LayerNorm(d_model)
        self.mlp = MLP(d_model, d_ff)

    def forward(self, x):
        x = x + self.attn(self.ln1(x))                   # pre-LN attention sub-block
        x = x + self.mlp(self.ln2(x))                    # pre-LN MLP sub-block
        return x

Block is §5.4 verbatim. Note the use of F.gelu (functional, stateless — §3.6) inside MLP, versus the nn.Module submodules (ln1, attn, …) that own parameters and so are registered by attribute assignment. Two separate LayerNorm instances — they learn different scales because they normalize different activations.

6.4 GPT: assembling the model

class GPT(nn.Module):
    def __init__(self, vocab, d_model=128, n_layers=3, n_heads=4, d_ff=512, max_T=256):
        super().__init__()
        self.max_T = max_T
        self.token_emb = nn.Embedding(vocab, d_model)
        self.pos_emb = nn.Embedding(max_T, d_model)
        self.blocks = nn.ModuleList(                      # MUST be ModuleList (§3.3)
            [Block(d_model, n_heads, d_ff) for _ in range(n_layers)])
        self.ln_f = LayerNorm(d_model)
        self.head = nn.Linear(d_model, vocab, bias=False)
        self.head.weight = self.token_emb.weight         # weight tying (§5.5)
        self.apply(self._init_weights)

    def _init_weights(self, m):
        if isinstance(m, nn.Linear):
            nn.init.normal_(m.weight, std=0.02)
            if m.bias is not None:
                nn.init.zeros_(m.bias)
        elif isinstance(m, nn.Embedding):
            nn.init.normal_(m.weight, std=0.02)

    def forward(self, idx):                              # idx: (B, T) ints
        B, T = idx.shape
        pos = torch.arange(T, device=idx.device)         # (T,)
        x = self.token_emb(idx) + self.pos_emb(pos)      # (B, T, D), pos broadcasts
        for block in self.blocks:
            x = block(x)
        x = self.ln_f(x)
        return self.head(x)                              # (B, T, vocab)

Everything from Part I shows up here. nn.ModuleList so the blocks register (a plain list would leave them untrained — §3.3). Weight tying by assigning self.head.weight = self.token_emb.weight — the two now share one (V, D) tensor, and since Linear computes x @ weight.T, that's the tied unembedding (§5.5). self.apply(fn) walks every submodule and runs fn on it — the standard initialization idiom, here normal(0, 0.02) for weights and zeros for biases. The forward is the §5.1 pipeline line for line.

Weight tying and the pos_emb device

Two subtleties. (1) Tying must use identical shapes: nn.Linear(D, V) has weight (V, D), and nn.Embedding(V, D) has weight (V, D) — they match, so the assignment shares storage cleanly. (2) torch.arange(T) defaults to CPU; on GPU it must be created with device=idx.device or the pos_emb lookup mixes devices and crashes. Both are exactly the kind of thing Part I trained you to catch.

6.5 Counting parameters as a sanity check

Before training, count: sum(p.numel() for p in model.parameters()). For our config on a vocab of 46 you should get 632,192 parameters. A number far from what you expect means a layer is the wrong size or — if it's suspiciously small — a sub-block isn't registered (the ModuleList/plain-list trap). This one line catches structural bugs before you waste a training run on them.

Tests before comparing: output shape equals input shape; and on an input where all tokens are identical, attention reduces to a uniform average over the visible prefix (so row t's output equals the mean of v[:t+1]). Finish with a comment: why does q @ k.transpose(-2,-1) produce all B·H score matrices at once, and which axis-1 op needs .contiguous()?

Order: project → view(B,T,H,hd) → transpose(1,2) → q@kᵀ/√hd → tril + masked_fill(-inf) → softmax(dim=-1) → @v → transpose(1,2).contiguous().view(B,T,D) → wo.
Three bugs to avoid: dividing by √D instead of √hd; masking after softmax; merging heads without .contiguous() (raises a stride error).
import math
class CausalSelfAttention(nn.Module):
    def __init__(self, d_model, n_heads):
        super().__init__()
        assert d_model % n_heads == 0
        self.n_heads = n_heads
        self.d_head = d_model // n_heads
        self.wq = nn.Linear(d_model, d_model, bias=False)
        self.wk = nn.Linear(d_model, d_model, bias=False)
        self.wv = nn.Linear(d_model, d_model, bias=False)
        self.wo = nn.Linear(d_model, d_model, bias=False)
    def forward(self, x):                                # (B, T, D)
        B, T, D = x.shape
        H, hd = self.n_heads, self.d_head
        q = self.wq(x).view(B, T, H, hd).transpose(1, 2)  # (B, H, T, hd)
        k = self.wk(x).view(B, T, H, hd).transpose(1, 2)
        v = self.wv(x).view(B, T, H, hd).transpose(1, 2)
        scores = (q @ k.transpose(-2, -1)) / math.sqrt(hd)  # (B, H, T, T)
        mask = torch.tril(torch.ones(T, T, dtype=torch.bool, device=x.device))
        scores = scores.masked_fill(~mask, float("-inf"))
        attn = F.softmax(scores, dim=-1)                    # (B, H, T, T)
        out = (attn @ v).transpose(1, 2).contiguous().view(B, T, D)  # (B,T,D)
        return self.wo(out)
# leading (B,H) are batch dims for @, so q @ kᵀ is a (T,hd)x(hd,T) matmul per
# (batch, head): all B·H score matrices in one op, no head loop.
# .contiguous() at the merge: transpose(1,2) makes the tensor non-contiguous
# and view() requires row-major memory.

        self.attn = CausalSelfAttention(d_model, n_heads)
